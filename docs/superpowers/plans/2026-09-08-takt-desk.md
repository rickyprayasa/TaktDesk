# Takt Desk Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the complete Takt Desk Android application in Flutter with an analog Braun / Dieter Rams industrial minimalist aesthetic, 60/120 FPS procedural Canvas instruments (Chrono Dial, Receipt Runner, Vinyl Deck), Drift SQLite persistence, low-latency foley audio, and plug-and-play Rive State Machine adapter readiness.

**Architecture:** Feature-first Clean Architecture with Riverpod for reactive state management. UI instruments implement a Dual-Engine pattern (Procedural Canvas as default active visual engine, Rive State Machine adapter ready for `.riv` drop-in). Gyroscope sensor parallax uses a low-pass filter, and dynamic ambient lighting adapts to local device time.

**Tech Stack:** Flutter 3.44+ (Dart 3.12+), `flutter_riverpod: ^2.5.1`, `drift: ^2.18.0`, `sqlite3_flutter_libs: ^0.5.24`, `path_provider: ^2.1.3`, `sensors_plus: ^5.0.1`, `rive: ^0.13.0`, `flutter_soloud: ^2.0.0` (with mock/fallback mode for headless testing).

## Global Constraints

- Android Target SDK 34+, Min SDK 24.
- Zero AI Slop: Strictly adhere to `TaktPalette` (#141517 Obsidian, #24272C Basalt, #EFECE6 Warm Bone, #1F2226 Thermal Carbon, #C85A32 Muted Ochre, #D99B59 Filament Amber, #4D5D4B Patina Olive). No neon gradients, no arbitrary rounded-3xl cards.
- Max 5 active tasks enforced in Thermal Task Runner.
- Sweep motion on Chrono Dial powered by Ticker / continuous animation.

```dart
import 'package:drift/drift.dart';
import 'package:drift/native.dart';
import 'package:path_provider/path_provider.dart';
import 'package:path/path.dart' as p;
import 'dart:io';

import 'tables/tasks_table.dart';
import 'tables/timer_sessions_table.dart';
import 'tables/desk_preferences_table.dart';

part 'app_database.g.dart';

@DriftDatabase(tables: [Tasks, TimerSessions, DeskPreferences])
class AppDatabase extends _$AppDatabase {
  AppDatabase() : super(_openConnection());
  AppDatabase.forTesting(super.e);

  @override
  int get schemaVersion => 1;

  Stream<List<Task>> watchActiveTasks() {
    return (select(tasks)
          ..where((t) => t.isArchived.equals(false))
          ..orderBy([(t) => OrderingTerm(expression: t.sortOrder)]))
        .watch();
  }

  Future<int> addTask(String title) async {
    final activeCount = await (select(tasks)..where((t) => t.isArchived.equals(false))).get();
    if (activeCount.length >= 5) {
      throw Exception('Maksimal 5 tugas aktif per batch kerja');
    }
    return into(tasks).insert(TasksCompanion.insert(
      title: title,
      sortOrder: Value(activeCount.length),
    ));
  }

  Future<void> toggleTaskCompleted(int id, bool completed) {
    return (update(tasks)..where((t) => t.id.equals(id))).write(
      TasksCompanion(
        isCompleted: Value(completed),
        completedAt: Value(completed ? DateTime.now() : null),
      ),
    );
  }

  Future<void> archiveTask(int id) {
    return (update(tasks)..where((t) => t.id.equals(id))).write(
      const TasksCompanion(isArchived: Value(true)),
    );
  }

  Future<int> recordTimerSession({
    required int targetMinutes,
    required int elapsedSeconds,
    required bool isFinished,
    required DateTime startedAt,
  }) {
    return into(timerSessions).insert(TimerSessionsCompanion.insert(
      targetMinutes: targetMinutes,
      elapsedSeconds: elapsedSeconds,
      isFinished: Value(isFinished),
      startedAt: Value(startedAt),
      finishedAt: Value(DateTime.now()),
    ));
  }
}

LazyDatabase _openConnection() {
  return LazyDatabase(() async {
    final dbFolder = await getApplicationDocumentsDirectory();
    final file = File(p.join(dbFolder.path, 'takt_desk.sqlite'));
    return NativeDatabase.createInBackground(file);
  });
}
```

```dart
class TimerState {
  final int targetMinutes;
  final double remainingSeconds;
  final bool isRunning;
  final int lastTickMinute;

  const TimerState({
    required this.targetMinutes,
    required this.remainingSeconds,
    required this.isRunning,
    this.lastTickMinute = 0,
  });

  double get progressPercent => targetMinutes > 0
      ? 1.0 - (remainingSeconds / (targetMinutes * 60.0)).clamp(0.0, 1.0)
      : 0.0;

  TimerState copyWith({
    int? targetMinutes,
    double? remainingSeconds,
    bool? isRunning,
    int? lastTickMinute,
  }) {
    return TimerState(
      targetMinutes: targetMinutes ?? this.targetMinutes,
      remainingSeconds: remainingSeconds ?? this.remainingSeconds,
      isRunning: isRunning ?? this.isRunning,
      lastTickMinute: lastTickMinute ?? this.lastTickMinute,
    );
  }
}
```

Create `lib/features/timer/presentation/chrono_dial_controller.dart`:
```dart
import 'package:flutter_riverpod/flutter_riverpod.dart';
import '../domain/timer_state.dart';
import '../../../core/haptics/haptic_service.dart';

class ChronoDialController extends StateNotifier<TimerState> {
  ChronoDialController()
      : super(const TimerState(
          targetMinutes: 25,
          remainingSeconds: 25 * 60.0,
          isRunning: false,
        ));

  void setTargetMinutes(int minutes) {
    if (state.isRunning) return;
    final clamped = minutes.clamp(1, 90);
    state = state.copyWith(
      targetMinutes: clamped,
      remainingSeconds: clamped * 60.0,
    );
    HapticService.dialTick();
  }

  void start() {
    state = state.copyWith(isRunning: true);
    HapticService.mechanicalPress();
  }

  void pause() {
    state = state.copyWith(isRunning: false);
    HapticService.mechanicalPress();
  }

  void reset() {
    state = state.copyWith(
      isRunning: false,
      remainingSeconds: state.targetMinutes * 60.0,
    );
  }

  void tick(double deltaSeconds) {
    if (!state.isRunning) return;
    final nextRemaining = (state.remainingSeconds - deltaSeconds).clamp(0.0, 90 * 60.0);
    final currentMinute = (nextRemaining / 60.0).ceil();

    if (currentMinute != state.lastTickMinute && nextRemaining > 0) {
      HapticService.dialTick();

    if (nextRemaining <= 0) {
      state = state.copyWith(remainingSeconds: 0, isRunning: false, lastTickMinute: 0);
      HapticService.mechanicalPress();
    } else {
      state = state.copyWith(
        remainingSeconds: nextRemaining,
        lastTickMinute: currentMinute,
      );
    }
  }
```

```dart
import 'dart:math' as math;
import 'package:flutter/material.dart';
import '../../../app/theme/takt_colors.dart';

class ChronoDialPainter extends CustomPainter {
  final double progressPercent;
  final int targetMinutes;
  final bool isRunning;

  ChronoDialPainter({
    required this.progressPercent,
    required this.targetMinutes,
    required this.isRunning,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final center = Offset(size.width / 2, size.height / 2);
    final radius = math.min(size.width, size.height) / 2 - 12;

    // 1. Basalt Slate housing bezel
    final housingPaint = Paint()
      ..color = TaktColors.basaltSlate
      ..style = PaintingStyle.fill;
    canvas.drawCircle(center, radius + 8, housingPaint);

    final borderPaint = Paint()
      ..color = TaktColors.slateBorder
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2.0;
    canvas.drawCircle(center, radius + 8, borderPaint);

    // 2. Amber halo ring if running
    if (isRunning) {
      final haloPaint = Paint()
        ..color = TaktColors.filamentAmber.withOpacity(0.25)
        ..style = PaintingStyle.stroke
        ..strokeWidth = 6.0
        ..maskFilter = const MaskFilter.blur(BlurStyle.normal, 8.0);
      canvas.drawCircle(center, radius + 4, haloPaint);
    }

    // 3. Dial Face
    final facePaint = Paint()
      ..color = const Color(0xFF1B1D21)
      ..style = PaintingStyle.fill;
    canvas.drawCircle(center, radius, facePaint);

    // 4. Tick Marks (60 ticks)
    final majorTickPaint = Paint()
      ..color = TaktColors.warmBone.withOpacity(0.8)
      ..strokeWidth = 2.0;
    final minorTickPaint = Paint()
      ..color = TaktColors.warmBone.withOpacity(0.3)
      ..strokeWidth = 1.0;

    for (int i = 0; i < 60; i++) {
      final angle = (i * 6) * math.pi / 180.0 - math.pi / 2;
      final isMajor = i % 5 == 0;
      final tickLength = isMajor ? 10.0 : 5.0;
      final start = Offset(
        center.dx + (radius - tickLength) * math.cos(angle),
        center.dy + (radius - tickLength) * math.sin(angle),
      );
      final end = Offset(
        center.dx + radius * math.cos(angle),
        center.dy + radius * math.sin(angle),
      );
      canvas.drawLine(start, end, isMajor ? majorTickPaint : minorTickPaint);
    }

    // 5. Sweep Hand (Muted Ochre)
    final handAngle = (progressPercent * 360.0) * math.pi / 180.0 - math.pi / 2;
    final handPaint = Paint()
      ..color = TaktColors.mutedOchre
      ..strokeWidth = 3.0
      ..strokeCap = StrokeCap.round;
    final handEnd = Offset(
      center.dx + (radius - 16) * math.cos(handAngle),
      center.dy + (radius - 16) * math.sin(handAngle),
    );
    canvas.drawLine(center, handEnd, handPaint);

    // Center Cap
    final capPaint = Paint()..color = TaktColors.mutedOchre;
    canvas.drawCircle(center, 5.0, capPaint);
  }

  @override
  bool shouldRepaint(covariant ChronoDialPainter oldDelegate) {
    return oldDelegate.progressPercent != progressPercent ||
        oldDelegate.targetMinutes != targetMinutes ||
        oldDelegate.isRunning != isRunning;
  }
}
```

```dart
class ReceiptRunnerController extends StateNotifier<ReceiptRunnerState> {

  bool get canAddTask => state.tasks.length < 5;

  void addTask(String title) {
    if (!canAddTask || title.trim().isEmpty) return;
    final newTask = ReceiptTask(
      id: DateTime.now().microsecondsSinceEpoch.toString(),
      title: title.trim(),
    );
    state = state.copyWith(tasks: [...state.tasks, newTask]);
    HapticService.mechanicalPress();
  }

    state = state.copyWith(
      tasks: state.tasks.map((t) => t.id == id ? t.copyWith(isDone: !t.isDone) : t).toList(),
    );
    HapticService.stampDone();
  }

  void updateTearProgress(double progress) {
    state = state.copyWith(tearDragProgress: progress.clamp(0.0, 1.0));
  }

  void tearTask(String id) {
    state = state.copyWith(
      tasks: state.tasks.where((t) => t.id != id).toList(),
      tearDragProgress: 0.0,
    );
    HapticService.paperTorn();
```

```dart
class ReceiptRunnerPainter extends CustomPainter {
  final double tearProgress;

  ReceiptRunnerPainter({this.tearProgress = 0.0});

  @override
  void paint(Canvas canvas, Size size) {
    // Draw serrated bottom edge of receipt paper
    final paperPaint = Paint()
      ..color = TaktColors.warmBone
      ..style = PaintingStyle.fill;

    final path = Path()..moveTo(0, 0);
    path.lineTo(size.width, 0);
    path.lineTo(size.width, size.height - 10);

    // Serration teeth
    const toothWidth = 8.0;
    const toothHeight = 6.0;
    final teethCount = (size.width / toothWidth).floor();

    for (int i = teethCount; i >= 0; i--) {
      final x = i * toothWidth;
      final y = (i % 2 == 0) ? size.height : size.height - toothHeight;
      path.lineTo(x, y);
    }
    path.close();

    canvas.drawPath(path, paperPaint);
  }

  @override
  bool shouldRepaint(covariant ReceiptRunnerPainter oldDelegate) {
    return oldDelegate.tearProgress != tearProgress;
  }
```

```dart
    // Turntable Platter

    canvas.drawCircle(center, radius + 4, platterPaint);

    // Vinyl Record with micro-grooves
    final vinylPaint = Paint()..color = const Color(0xFF0F1012);

    final groovePaint = Paint()
      ..color = Colors.white.withOpacity(0.04)
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1.0;
    for (double r = radius * 0.4; r < radius; r += 6.0) {
      canvas.drawCircle(center, r, groovePaint);
    }

    final labelPaint = Paint()..color = TaktColors.mutedOchre;
    canvas.drawCircle(center, radius * 0.3, labelPaint);

    canvas.drawCircle(center, 4.0, spindlePaint);

    // Tonearm
    final tonearmBase = Offset(size.width * 0.88, size.height * 0.2);
    final armTarget = needleOnRecord ? center + Offset(radius * 0.5, 0) : tonearmBase + const Offset(-10, 50);

    final armPaint = Paint()

    canvas.drawCircle(tonearmBase, 10.0, pivotPaint);

    return oldDelegate.rotationAngle != rotationAngle ||
```

```dart
import 'package:flutter_riverpod/flutter_riverpod.dart';

class ParallaxState {
  final double tiltX;
  final double tiltY;

  const ParallaxState({this.tiltX = 0.0, this.tiltY = 0.0});
}

class ParallaxController extends StateNotifier<ParallaxState> {
  ParallaxController() : super(const ParallaxState());

  static const double _alpha = 0.15; // Low-pass filter smoothing coefficient

  void updateSensorInput({required double rawX, required double rawY}) {
    // Clamp to max 3 degrees (approx 0.05 radians)
    final targetX = (rawX * 0.01).clamp(-0.05, 0.05);
    final targetY = (rawY * 0.01).clamp(-0.05, 0.05);

    final smoothX = _alpha * targetX + (1 - _alpha) * state.tiltX;
    final smoothY = _alpha * targetY + (1 - _alpha) * state.tiltY;

    state = ParallaxState(tiltX: smoothX, tiltY: smoothY);
  }
}

final parallaxProvider = StateNotifierProvider<ParallaxController, ParallaxState>((ref) {
  return ParallaxController();
});
```

```dart
class AmbientLightingLayer extends StatelessWidget {
  final Widget child;
  final DateTime? currentTime;

  const AmbientLightingLayer({
    super.key,
    required this.child,
    this.currentTime,
  });

  @override
  Widget build(BuildContext context) {
    final now = currentTime ?? DateTime.now();
    final hour = now.hour;

    // Morning (6-11), Afternoon (11-17), Night (17-6)
    Color lightTint;
    Alignment lightOrigin;

    if (hour >= 6 && hour < 11) {
      lightTint = const Color(0x10FFFFFF); // Cool morning daylight
      lightOrigin = Alignment.topLeft;
    } else if (hour >= 11 && hour < 17) {
      lightTint = const Color(0x15FFD6A5); // Warm afternoon sunlight
      lightOrigin = Alignment.topCenter;
    } else {
      lightTint = TaktColors.filamentAmber.withOpacity(0.08); // Filament night lamp
      lightOrigin = Alignment.topRight;
    }

    return Stack(
      children: [
        child,
        IgnorePointer(
          child: Container(
            decoration: BoxDecoration(
              gradient: RadialGradient(
                center: lightOrigin,
                radius: 1.5,
                colors: [lightTint, Colors.transparent],
              ),
            ),
          ),
        ),
      ],
    );
  }
}
```
